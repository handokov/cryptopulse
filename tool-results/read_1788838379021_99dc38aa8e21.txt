   205→   205→function DonutTooltip({ active, payload }: { active?: boolean; payload?: Array<{ payload?: DonutDatum }> }) {
   206→   206→  if (!active || !payload || payload.length === 0) return null;
   207→   207→  const datum = payload[0]?.payload;
   208→   208→  if (!datum) return null;
   209→   209→  return (
   210→   210→    <div className="rounded-md border border-border bg-card px-2 py-1 text-xs">
   211→   211→      <span className="font-semibold text-foreground">{datum.name}</span>{" "}
   212→   212→      <span className="tnum text-muted-foreground">{fmtCompactUsd(datum.value)}</span>
   213→   213→    </div>
   214→   214→  );
   215→   215→}

   236→   236→export function PortfolioSection() {
   237→   237→  const t = useTranslations("portfolio");
   238→   238→  const tAuth = useTranslations("auth");
   239→   239→  const status = useAuthStore((s) => s.status);
   240→   240→  const setDialogOpen = useAuthStore((s) => s.setDialogOpen);
   241→   241→  const locale = useLocaleStore((s) => s.locale);
   242→   242→
   243→   243→  /* portfolio data */
   244→   244→  const [data, setData] = useState<PortfolioPayload | null>(null);
   245→   245→  const [loading, setLoading] = useState(true);
   246→   246→  const [error, setError] = useState(false);
   247→   247→
   248→   248→  /* add / edit dialog */
   249→   249→  const [dialogMode, setDialogMode] = useState<"add" | "edit" | null>(null);
   250→   250→  const [editing, setEditing] = useState<Holding | null>(null);
   251→   251→  const [coin, setCoin] = useState<CoinOption | null>(null);
   252→   252→  const [quantity, setQuantity] = useState("");
   253→   253→  const [price, setPrice] = useState("");
   254→   254→  const [priceTouched, setPriceTouched] = useState(false);
   255→   255→  const [date, setDate] = useState("");
   256→   256→  const [coinPickerOpen, setCoinPickerOpen] = useState(false);
   257→   257→  const [coinOptions, setCoinOptions] = useState<Top100Coin[]>([]);
   258→   258→  const [saving, setSaving] = useState(false);
   259→   259→  const [formError, setFormError] = useState<"validation" | "error" | null>(null);
   260→   260→
   261→   261→  /* delete dialog */
   262→   262→  const [deleting, setDeleting] = useState<Holding | null>(null);
   263→   263→  const [deletingPending, setDeletingPending] = useState(false);
   264→   264→  const [deleteError, setDeleteError] = useState(false);
   265→   265→
   266→   266→  const load = useCallback(async () => {
   267→   267→    setLoading(true);
   268→   268→    try {
   269→   269→      const res = await fetch("/api/portfolio", { cache: "no-store" });
   270→   270→      if (!res.ok) throw new Error(`portfolio fetch failed: ${res.status}`);
   271→   271→      setData((await res.json()) as PortfolioPayload);
   272→   272→      setError(false);
   273→   273→    } catch {
   274→   274→      setError(true);
   275→   275→    } finally {
   276→   276→      setLoading(false);
   277→   277→    }
   278→   278→  }, []);
   279→   279→
   280→   280→  /* (re)load whenever the session flips to authenticated */
   281→   281→  useEffect(() => {
   282→   282→    if (status === "authenticated") void load();
   283→   283→  }, [status, load]);
   284→   284→
   285→   285→  /* fetch the top-100 coin universe the first time the add/edit dialog opens */
   286→   286→  useEffect(() => {
   287→   287→    if (!dialogMode || coinOptions.length > 0) return;
   288→   288→    void (async () => {
   289→   289→      try {
   290→   290→        const res = await fetch("/api/market/top100", { cache: "no-store" });
   291→   291→        if (!res.ok) return;
   292→   292→        const json = (await res.json()) as Top100Payload;
   293→   293→        setCoinOptions(json.coins ?? []);
   294→   294→      } catch {
   295→   295→        /* picker stays empty → CommandEmpty shows */
   296→   296→      }
   297→   297→    })();
   298→   298→  }, [dialogMode, coinOptions.length]);
   299→   299→
   300→   300→  const holdings = useMemo(() => data?.holdings ?? [], [data]);
   301→   301→  const computed = data?.computed ?? null;
   302→   302→
   303→   303→  const valued = useMemo(
   304→   304→    () => holdings.filter((h): h is Holding & { currentValue: number } => h.currentValue != null),
   305→   305→    [holdings]
   306→   306→  );
   307→   307→  const withPnl = useMemo(
   308→   308→    () => holdings.filter((h): h is Holding & { pnlPct: number } => h.pnlPct != null),
   309→   309→    [holdings]
   310→   310→  );
   311→   311→  const best = useMemo(
   312→   312→    () => (withPnl.length ? withPnl.reduce((a, b) => (b.pnlPct > a.pnlPct ? b : a)) : null),
   313→   313→    [withPnl]
   314→   314→  );
   315→   315→  const worst = useMemo(
   316→   316→    () => (withPnl.length ? withPnl.reduce((a, b) => (b.pnlPct < a.pnlPct ? b : a)) : null),
   317→   317→    [withPnl]
   318→   318→  );
   319→   319→  const uniqueSymbols = useMemo(() => new Set(holdings.map((h) => h.coinId)).size, [holdings]);
   320→   320→
   321→   321→  const donutData = useMemo<DonutDatum[]>(
   322→   322→    () => valued.map((h) => ({ name: h.symbol, value: h.currentValue, weight: h.weight ?? 0 })),
   323→   323→    [valued]
   324→   324→  );
   325→   325→  const barData = useMemo<PnlDatum[]>(
   326→   326→    () => valued.map((h) => ({ symbol: h.symbol, pnlPct: h.pnlPct ?? 0 })),
   327→   327→    [valued]
   328→   328→  );
   329→   329→
   330→   330→  /* insight lines — only built when data is present */
   331→   331→  const insightLines = useMemo(() => {
   332→   332→    const lines: Array<{ color: string; text: string }> = [];
   333→   333→    if (!computed) return lines;
   334→   334→    if (computed.totalPnlPct != null) {
   335→   335→      lines.push({
   336→   336→        color: computed.totalPnlPct >= 0 ? "bg-primary" : "bg-destructive",
   337→   337→        text:
   338→   338→          computed.totalPnlPct >= 0
   339→   339→            ? t("insightOverallPositive", { pct: fmtPct(computed.totalPnlPct) })
   340→   340→            : t("insightOverallNegative", { pct: fmtPct(computed.totalPnlPct) }),
   341→   341→      });
   342→   342→    }
   343→   343→    if (computed.positionCount > 0) {
   344→   344→      lines.push({
   345→   345→        color: "bg-amber-400",
   346→   346→        text: t("insightProfit", { count: computed.profitableCount, total: computed.positionCount }),
   347→   347→      });
   348→   348→    }
   349→   349→    if (computed.topHolding) {
   350→   350→      lines.push(
   351→   351→        computed.topHolding.weight > 40
   352→   352→          ? {
   353→   353→              color: "bg-amber-400",
   354→   354→              text: t("insightConcentrated", {
   355→   355→                name: computed.topHolding.symbol,
   356→   356→                pct: fmtPct(computed.topHolding.weight),
   357→   357→              }),
   358→   358→            }
   359→   359→          : {
   360→   360→              color: "bg-primary",
   361→   361→              text: t("insightTop", {
   362→   362→                name: computed.topHolding.symbol,
   363→   363→                pct: fmtPct(computed.topHolding.weight),
   364→   364→              }),
   365→   365→            }
   366→   366→      );
   367→   367→    }
   368→   368→    if (computed.positionCount >= 3) {
   369→   369→      lines.push({
   370→   370→        color: "bg-amber-400",
   371→   371→        text: t("insightDiversified", { count: computed.positionCount, assets: uniqueSymbols }),
   372→   372→      });
   373→   373→    }
   374→   374→    if (best) {
   375→   375→      lines.push({ color: "bg-primary", text: t("insightBest", { name: best.symbol, pct: fmtPct(best.pnlPct) }) });
   376→   376→    }
   377→   377→    if (worst) {
   378→   378→      lines.push({
   379→   379→        color: worst.pnlPct < 0 ? "bg-destructive" : "bg-primary",
   380→   380→        text: t("insightWorst", { name: worst.symbol, pct: fmtPct(worst.pnlPct) }),
   381→   381→      });
   382→   382→    }
   383→   383→    return lines;
   384→   384→  }, [computed, uniqueSymbols, best, worst, t]);
   385→   385→
   386→   386→  /* ---- form helpers ---- */
   387→   387→
   388→   388→  const resetForm = useCallback(() => {
   389→   389→    setCoin(null);
   390→   390→    setQuantity("");
   391→   391→    setPrice("");
   392→   392→    setPriceTouched(false);
   393→   393→    setDate("");
   394→   394→    setCoinPickerOpen(false);
   395→   395→    setFormError(null);
   396→   396→    setSaving(false);
   397→   397→  }, []);

   400→   400→    setEditing(null);
   401→   401→    setDialogMode("add");
   402→   402→    resetForm();
   403→   403→    setDate(new Date().toISOString().slice(0, 10));
   404→   404→  }, [resetForm]);
   405→   405→
   406→   406→  const openEdit = useCallback(
   407→   407→    (h: Holding) => {
   408→   408→      setEditing(h);
   409→   409→      setDialogMode("edit");
   410→   410→      resetForm();
   411→   411→      setCoin({ id: h.coinId, symbol: h.symbol, name: h.name, image: h.image, price: h.currentPrice ?? h.purchasePrice });
   412→   412→      setQuantity(String(h.quantity));
   413→   413→      setPrice(String(h.purchasePrice));
   414→   414→      setPriceTouched(true);
   415→   415→      setDate(h.purchaseDate.slice(0, 10));
   416→   416→    },
   417→   417→    [resetForm]
   418→   418→  );
   419→   419→
   420→   420→  const closeForm = useCallback(() => {
   421→   421→    setDialogMode(null);
   422→   422→    setEditing(null);
   423→   423→    resetForm();
   424→   424→  }, [resetForm]);
   425→   425→
   426→   426→  const pickCoin = useCallback(
   427→   427→    (c: Top100Coin) => {
   428→   428→      setCoin({ id: c.id, symbol: c.symbol, name: c.name, image: c.image, price: c.price });
   429→   429→      if (dialogMode === "add" && !priceTouched && c.price > 0) {
   430→   430→        setPrice(String(Math.round(c.price * 1e6) / 1e6));
   431→   431→      }
   432→   432→      setCoinPickerOpen(false);
   433→   433→    },
   434→   434→    [dialogMode, priceTouched]
   435→   435→  );
   436→   436→
   437→   437→  const saveHolding = useCallback(async () => {
   438→   438→    if (!dialogMode || saving) return;
   439→   439→    const q = Number(quantity);
   440→   440→    const p = Number(price);
   441→   441→    if (!coin || !quantity || !price || !date || !Number.isFinite(q) || q <= 0 || !Number.isFinite(p) || p <= 0) {
   442→   442→      setFormError("validation");
   443→   443→      return;
   444→   444→    }
   445→   445→    setSaving(true);
   446→   446→    setFormError(null);
   447→   447→    const isEdit = dialogMode === "edit" && editing != null;
   448→   448→    try {
   449→   449→      const res = await fetch(isEdit ? `/api/portfolio/${(editing as Holding).id}` : "/api/portfolio", {
   450→   450→        method: isEdit ? "PATCH" : "POST",
   451→   451→        headers: { "Content-Type": "application/json" },
   452→   452→        body: JSON.stringify(
   453→   453→          isEdit
   454→   454→            ? { quantity: q, purchasePrice: p, purchaseDate: new Date(date).toISOString() }
   455→   455→            : {
   456→   456→                coinId: coin.id,
   457→   457→                symbol: coin.symbol,
   458→   458→                name: coin.name,
   459→   459→                image: coin.image ?? undefined,
   460→   460→                quantity: q,
   461→   461→                purchasePrice: p,
   462→   462→                purchaseDate: new Date(date).toISOString(),
   463→   463→              }
   464→   464→        ),
   465→   465→      });
   466→   466→      if (res.ok) {
   467→   467→        closeForm();
   468→   468→        void load();
   469→   469→      } else {
   470→   470→        setFormError("error");
   471→   471→      }
   472→   472→    } catch {
   473→   473→      setFormError("error");
   474→   474→    } finally {
   475→   475→      setSaving(false);
   476→   476→    }
   477→   477→  }, [dialogMode, saving, quantity, price, date, coin, editing, closeForm, load]);
   478→   478→
   479→   479→  const confirmDelete = useCallback(async () => {
   480→   480→    if (!deleting || deletingPending) return;
   481→   481→    setDeletingPending(true);
   482→   482→    setDeleteError(false);
   483→   483→    try {
   484→   484→      const res = await fetch(`/api/portfolio/${deleting.id}`, { method: "DELETE" });
   485→   485→      if (!res.ok) throw new Error("delete failed");
   486→   486→      setDeleting(null);
   487→   487→      void load();
   488→   488→    } catch {
